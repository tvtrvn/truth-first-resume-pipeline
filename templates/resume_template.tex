% --- FONT STACK (ATS text-extraction hardening) ---
% Was: fontenc[T1] + lmodern (Type1). Two extraction bugs found empirically:
%  1. LIGATURES: "workflows"/"staff"/"certifications" extracted as Unicode
%     presentation forms (U+FB00-FB02) -- exact-string ATS keyword matchers
%     miss those tokens. Ligatures=NoCommon kills fi/fl/ff ligation entirely.
%  2. KERN SPLITS: OTF kern pairs at heading size ("Ta" in "Taylor") extract
%     with a spurious space ("T aylor") in pypdf-class extractors. Kerning=Off
%     eliminates it.
% Visual cost at 10pt is negligible; extraction fidelity is the priority.
% Gate: every render must pass the text-extraction check in
% latex-to-pdf-renderer (no U+FB00-FB06, no U+FFFD, name/email/keywords survive).
\usepackage{fontspec}

  BoldFont=lmroman10-bold.otf,
  ItalicFont=lmroman10-italic.otf,
  BoldItalicFont=lmroman10-bolditalic.otf,

% --- Robust Unicode punctuation under Tectonic (XeTeX) + legacy T1/lmodern fonts ---
% Raw Unicode symbols in tailored content would otherwise land in wrong T1 font
% slots and render as garbage -- e.g. a literal middle-dot "·" (U+00B7) prints as
% a u-with-accent glyph. Map the symbols the tailor commonly emits to the correct
% LaTeX commands so content can safely contain them. Extend this list if a new
% symbol ever renders wrong. (Fixes the "·-renders-as-u-with-accent" bug.)
\usepackage{newunicodechar}
\newunicodechar{·}{\textperiodcentered}
\newunicodechar{–}{\textendash}
\newunicodechar{—}{\textemdash}
\newunicodechar{•}{\textbullet}
\newunicodechar{×}{\ensuremath{\times}}
\newunicodechar{≥}{\ensuremath{\geq}}
\newunicodechar{≤}{\ensuremath{\leq}}
\newunicodechar{→}{\ensuremath{\rightarrow}}
\newunicodechar{…}{\ldots}

% ============================================================================
%  SPACING CONTRACT -- LOCKED. Read before touching anything below.
%  ----------------------------------------------------------------------------
%  These vertical-spacing values are the SINGLE SOURCE OF TRUTH for the resume's
%  rhythm. They are deliberately POSITIVE and UNIFORM so every section and bullet
%  gets the same breathing room and lines never collide. They were validated to
%  produce a clean, legible 1-page resume (~51 content lines, page_metrics PASS).
%
%  HARD RULES (enforced by resume-tailor / latex-to-pdf-renderer / output-packager):
%   1. NEVER reduce these values to force a 1-page fit. NEVER use negative or
%      sub-1pt itemsep/parsep/topsep, and NEVER shrink the margin below 0.5in or
%      the font below 10pt. Crushed/negative spacing is what makes lines overlap
%      and the page look squished -- the exact regression this contract prevents.
%   2. The ONLY lever for a >1-page overflow is trimming CONTENT: drop the weakest
%      bullet or project, then tighten wording (~22 words/bullet). Re-measure with
%      page_metrics.py and repeat. Fit the content to the spacing, never the
%      spacing to the content.
%   3. To FILL a sparse page, ADD truthful content (see tailor step 12) -- do not
%      inflate spacing or font either.
%  Any agent that edits the four values below to hit a page target is in breach.
% ============================================================================
